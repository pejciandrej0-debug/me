(()=>{'use strict';
class FixedClock{constructor(hz=60,maxSteps=5){this.step=1/hz;this.maxSteps=maxSteps;this.acc=0;this.alpha=0;this.steps=0;}reset(){this.acc=this.alpha=this.steps=0;}advance(elapsed,tick){this.acc=Math.min(this.acc+Math.max(0,Math.min(.05,Number(elapsed)||0)),this.step*this.maxSteps);this.steps=0;while(this.acc+1e-10>=this.step&&this.steps<this.maxSteps){tick(this.step);this.acc-=this.step;this.steps++;}this.alpha=Math.max(0,Math.min(1,this.acc/this.step));return this.alpha;}}
class SpatialGrid{constructor(cell=256){this.cell=cell;this.buckets=new Map();}key(x,y){return (Math.floor(x/this.cell)+32768)*65536+Math.floor(y/this.cell)+32768;}rebuild(items,x='x',y='y'){for(const b of this.buckets.values())b.length=0;for(const a of items){const key=this.key(a[x],a[y]);let b=this.buckets.get(key);if(!b)this.buckets.set(key,b=[]);b.push(a);}this.x=x;this.y=y;}query(x0,y0,x1,y1,out=[]){out.length=0;const C=this.cell;for(let x=Math.floor(x0/C);x<=Math.floor(x1/C);x++)for(let y=Math.floor(y0/C);y<=Math.floor(y1/C);y++){const b=this.buckets.get((x+32768)*65536+y+32768);if(b)for(const a of b)if(a[this.x]>=x0&&a[this.x]<=x1&&a[this.y]>=y0&&a[this.y]<=y1)out.push(a);}return out;}}
class Pool{constructor(make,limit=128){this.make=make;this.limit=limit;this.free=[];}take(){return this.free.pop()||this.make();}release(item){if(this.free.length<this.limit)this.free.push(item);}}
function repellerDecision(random,assist=false){return random()<(assist?.8:.65);}
class PathDistance{
 constructor(paths,cell=128){this.cell=cell;this.segments=[];this.grid=new Map();for(const p of paths)for(let i=1;i<p.length;i++){const a=p[i-1],b=p[i],id=this.segments.length;this.segments.push([a,b]);for(let x=Math.floor(Math.min(a[0],b[0])/cell);x<=Math.floor(Math.max(a[0],b[0])/cell);x++)for(let y=Math.floor(Math.min(a[1],b[1])/cell);y<=Math.floor(Math.max(a[1],b[1])/cell);y++){const k=(x+32768)*65536+y+32768;let q=this.grid.get(k);if(!q)this.grid.set(k,q=[]);q.push(id);}}this.seen=new Uint32Array(this.segments.length);this.stamp=0;}
 distance(x,y){const C=this.cell,cx=Math.floor(x/C),cy=Math.floor(y/C),stamp=++this.stamp;let best=Infinity;const visit=id=>{if(this.seen[id]===stamp)return;this.seen[id]=stamp;const [a,b]=this.segments[id],dx=b[0]-a[0],dy=b[1]-a[1],t=Math.max(0,Math.min(1,((x-a[0])*dx+(y-a[1])*dy)/(dx*dx+dy*dy||1)));best=Math.min(best,Math.hypot(x-a[0]-t*dx,y-a[1]-t*dy));};for(let r=0;r<=3;r++){for(let i=-r;i<=r;i++)for(let j=-r;j<=r;j++){if(r&&Math.abs(i)!==r&&Math.abs(j)!==r)continue;const q=this.grid.get((cx+i+32768)*65536+cy+j+32768);if(q)for(const id of q)visit(id);}const edge=Math.min(x-(cx-r)*C,(cx+r+1)*C-x,y-(cy-r)*C,(cy+r+1)*C-y);if(best<=edge)return best;}for(let id=0;id<this.segments.length;id++)visit(id);return best;}
}
function nearestSample(x,y,points,lengths,limit,at,step=5){let best=Infinity,result=0;const max=Math.floor(lengths[limit]/step)*step;for(let i=1;i<=limit;i++){const a=points[i-1],b=points[i],dx=b[0]-a[0],dy=b[1]-a[1],u=Math.max(0,Math.min(1,((x-a[0])*dx+(y-a[1])*dy)/(dx*dx+dy*dy||1))),s=Math.floor((lengths[i-1]+u*(lengths[i]-lengths[i-1]))/step)*step;for(let k=0;k<2;k++){const candidate=Math.max(0,Math.min(max,s+k*step)),p=at(candidate,0),d=Math.hypot(p[0]-x,p[1]-y);if(d<best||d===best&&candidate<result){best=d;result=candidate;}}}return result;}
