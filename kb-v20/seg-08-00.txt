  float u=clamp((s.x+float(c))/uCascN,float(c)/uCascN+m,float(c+1)/uCascN-m);
  vec2 uv=vec2(u,clamp(s.y,uShadowTexel.y*2.5,1.-uShadowTexel.y*2.5));
  float slopeBias=clamp(1.-abs(dot(N,normalize(uLDir))),0.,1.)*(c==0?.0007:.0014);
  float ref=s.z-(c==0?.00045:.0011)-slopeBias;
#if SHADOW_TAPS>1
  float rot=h21(floor(uv/uShadowTexel))*6.2831853;
  float spread=c==0?1.85:1.55;
  float sh=0.;
  for(int i=0;i<SHADOW_TAPS;i++){
    float r=sqrt((float(i)+.5)/float(SHADOW_TAPS))*spread;
    float a=float(i)*2.3999632+rot;
    vec2 o=vec2(cos(a),sin(a))*r*uShadowTexel;
    sh+=texture(uSunShadow,vec3(uv+o,ref));
  }
  return sh/float(SHADOW_TAPS);
#else
  return texture(uSunShadow,vec3(uv,ref));
#endif
}
#endif

float sunShadow(vec3 P,vec3 N,float d){
#ifdef SUN_SHADOW
  if(d>uCasc.y||uLCol.x+uLCol.y+uLCol.z<.001)return 1.;
  float sh;
  if(uCascN>1.5){
    if(d<uCasc.x*.82)sh=sampleCascade(0,P,N);
    else if(d<uCasc.x){float s0=sampleCascade(0,P,N),s1=sampleCascade(1,P,N);sh=mix(s0,s1,smoothstep(uCasc.x*.82,uCasc.x,d));}
    else sh=sampleCascade(1,P,N);
  }else sh=sampleCascade(0,P,N);
  return mix(sh,1.,smoothstep(.84,1.,d/uCasc.y));
#else
  return 1.;
#endif
}

float spotShadow(vec3 P,vec3 N){
#ifdef SPOT_SHADOW
  if(uSpotOn<.5)return 1.;
  vec4 q=uSpotVP*vec4(P+N*.055,1.);if(q.w<=0.)return 1.;vec3 s=q.xyz/q.w*.5+.5;
  if(s.x<0.||s.x>1.||s.y<0.||s.y>1.||s.z>1.)return 1.;
  float ref=s.z-.00075;float sh=0.;
  float rot=ign(gl_FragCoord.xy)*6.2831853;
  for(int i=0;i<8;i++){
    float r=sqrt((float(i)+.5)/8.)*1.75;
    float a=float(i)*2.3999632+rot;
    sh+=texture(uSpotShadow,vec3(s.xy+vec2(cos(a),sin(a))*r*uSpotTexel,ref));
  }
  return sh*.125;
#else
  return 1.;
#endif
}

vec4 tex1(float L,vec3 p,vec3 N,vec3 w,float sc,out vec3 nrm,out float rm){
#ifdef KB_ARCH_HIGH
  if((L==1.||L==2.||L==4.||L==6.)&&distance(p,uEye)<29.){
    vec3 U=w.y>max(w.x,w.z)?vec3(1,0,0):(w.x>w.z?vec3(0,0,1):vec3(1,0,0));
    vec3 B=w.y>max(w.x,w.z)?vec3(0,0,1):vec3(0,1,0);
    vec3 E=normalize(uEye-p);
    vec2 tc=vec2(dot(p,U),dot(p,B))*sc, gx=dFdx(tc), gy=dFdy(tc);
    vec2 delta=vec2(dot(E,U),dot(E,B))/max(.26,abs(dot(E,N)))*(L==4.?.019:.053)*sc;
    delta*=1.-smoothstep(15.,29.,distance(p,uEye));
    float depth=0., prev=0., diff=0.;vec2 q=tc;
    for(int j=0;j<10;j++){
      diff=depth-(1.-textureGrad(uTexA,vec3(q,L),gx,gy).a);
      if(diff>=0.)break;
      prev=diff;q-=delta*.111111;depth+=.111111;
    }
    q+=delta*.111111*clamp(diff/max(.001,diff-prev),0.,1.);
    vec4 normalR=textureGrad(uTexN,vec3(q,L),gx,gy);
    vec3 un=normalize(U-N*dot(U,N)),bn=normalize(B-N*dot(B,N)-un*dot(B,un));
    nrm=normalize(N*max(.25,normalR.b*2.-1.)+un*(normalR.r*2.-1.)+bn*(normalR.g*2.-1.));
    rm=normalR.a;return textureGrad(uTexA,vec3(q,L),gx,gy);
  }
#endif
#ifdef TRIPLANAR
  vec4 a=texture(uTexA,vec3(p.zy*sc,L)),b=texture(uTexA,vec3(p.xz*sc,L)),c=texture(uTexA,vec3(p.xy*sc,L));
  vec4 r=a*w.x+b*w.y+c*w.z;
  #ifdef NORMALMAP
  vec4 nx4=texture(uTexN,vec3(p.zy*sc,L)),ny4=texture(uTexN,vec3(p.xz*sc,L)),nz4=texture(uTexN,vec3(p.xy*sc,L));
  vec3 tx=nx4.xyz*2.-1.,ty=ny4.xyz*2.-1.,tz=nz4.xyz*2.-1.;
  tx=vec3(tx.xy+N.zy,N.x);ty=vec3(ty.xy+N.xz,N.y);tz=vec3(tz.xy+N.xy,N.z);
  nrm=normalize(tx.zyx*w.x+ty.xzy*w.y+tz.xyz*w.z);
  rm=nx4.a*w.x+ny4.a*w.y+nz4.a*w.z;
  #else
  nrm=N;rm=.5;
  #endif
  return r;
#else
  vec2 uv=w.y>.5?p.xz:(w.x>w.z?p.zy:p.xy);
  nrm=N;rm=.5;return texture(uTexA,vec3(uv*sc,L));
#endif
}

vec3 groundColor(vec2 xz,vec4 m,float slope){
  vec3 mead,forest,park,dry;
  if(uSeason<.5){mead=vec3(.34,.51,.22);forest=vec3(.14,.31,.15);park=vec3(.27,.51,.21);dry=vec3(.48,.44,.26);}
  else if(uSeason<1.5){mead=vec3(.32,.44,.20);forest=vec3(.13,.27,.14);park=vec3(.24,.42,.18);dry=vec3(.50,.45,.25);}
  else if(uSeason<2.5){mead=vec3(.50,.41,.21);forest=vec3(.36,.27,.14);park=vec3(.43,.37,.19);dry=vec3(.46,.34,.20);}
  else{mead=vec3(.82,.85,.86);forest=vec3(.66,.72,.71);park=vec3(.84,.87,.88);dry=vec3(.70,.73,.74);}
  float n1=vnoise(xz*.038),n2=vnoise(xz*.26+11.),n3=vnoise(xz*1.85+5.);
  vec3 c=mix(mead,dry,smoothstep(.40,.78,n1)*.62);
  c=mix(c,park,m.g);
  // Sumska prostirka (lisce/humus) u sumskim delovima Kalac brda
  vec3 litter=uSeason>2.5?vec3(.58,.62,.62):vec3(.23,.18,.11);
  c=mix(c,mix(forest*(.78+.42*n2),litter,smoothstep(.42,.74,n2)*.48),m.r);
  c*=.83+.30*n2+.12*(n3-.5);
  // Prirodni prelaz pored ulica i dvorista (plavi kanal maske terena)
  vec3 verge=uSeason>2.5?vec3(.68,.70,.72):vec3(.42,.39,.31);
  c=mix(c,verge,m.b*smoothstep(.35,.72,n2)*.38);
  // Strme padine Kalac brda: slojevi zemlje i krecnjackog kamena
  vec3 soil=uSeason>2.5?vec3(.55,.57,.59):vec3(.38,.29,.20);
  vec3 rock=uSeason>2.5?vec3(.52,.55,.58):vec3(.47,.45,.41);
  c=mix(c,soil,smoothstep(.13,.28,slope)*.72);
  c=mix(c,rock,smoothstep(.27,.46,slope)*.78*(.8+.35*n3));
  if(uSeason<.5){vec2 g=floor(xz*2.6);float f=step(.94,h21(g)),k=h21(g+7.);c=mix(c,k<.5?vec3(.95,.95,.88):vec3(.95,.80,.20),f*.85*(1.-m.r)*smoothstep(.45,.62,n1));}
  c=mix(vec3(dot(c,vec3(.33,.5,.17))),c,.86)*.94;
  return c;
}

// A window is an aperture into a finite room. Test nearer furniture boxes before the enclosing walls.
float roomBox(vec3 o,vec3 invRay,vec3 lo,vec3 hi){
  vec3 a=(lo-o)*invRay,b=(hi-o)*invRay,n=min(a,b),f=max(a,b);
  float nearT=max(max(n.x,n.y),n.z),farT=min(min(f.x,f.y),f.z);
  return nearT>0.&&nearT<farT?nearT:1e4;
}
vec3 interiorRoom(vec2 w,vec2 halfWindow,vec3 N,vec3 V,float seed,out float glow){
  vec3 U=normalize(cross(vec3(0,1,0),N));vec3 d=normalize(vec3(-dot(V,U),-V.y,max(.065,dot(V,N))));
  vec3 iv=vec3(d.x<0.?-1.:1.,d.y<0.?-1.:1.,1.)/max(abs(d),vec3(.001));
  vec3 o=vec3(w,0.),hi=vec3(1.6+seed*.55,1.38,2.8+seed),lo=vec3(-hi.x,-1.48,0.);
  vec3 f=max((lo-o)*iv,(hi-o)*iv);float t=min(f.x,min(f.y,f.z));vec3 hit=o+t*d;
  vec3 plaster=mix(vec3(.42,.38,.30),vec3(.39,.45,.41),seed);
  vec3 c= f.y<t+.001?(d.y>0.?vec3(.70,.68,.59):vec3(.23,.145,.09)*(.78+.22*step(.06,fract(hit.x*4.)))):plaster*(f.x<t+.001?.71:1.);
  float frame=step(abs(hit.x+.52),.45)*step(abs(hit.y-.08),.33)*step(f.z,t+.001);
  c=mix(c,vec3(.22,.28,.23),frame);
  float shelf=roomBox(o,iv,vec3(.54,-1.48,1.68),vec3(1.33,-.33,2.13));
  float chair=roomBox(o,iv,vec3(-1.18,-1.24,1.05),vec3(-.43,-.65,1.69));
  if(min(shelf,chair)<t){hit=o+d*min(shelf,chair);c=shelf<chair?vec3(.23,.14,.085)*(.72+.28*step(.06,fract(hit.y*4.))):vec3(.26,.30,.28);}
  float drape=smoothstep(halfWindow.x*.60,halfWindow.x*.95,abs(w.x+.06*d.x))*(.35+.65*seed);
  c=mix(c,vec3(.58,.50,.37)*(.82+.18*cos(w.x*77.)),drape);
  glow=(1.-smoothstep(.3,3.6,length(hit-vec3(.28,1.02,1.6))))*.6+.24;
  return c;
}
void facade(inout vec3 base,inout vec3 emis,inout float glass,inout float rough,inout float spec,inout vec3 Nm,vec3 N,vec3 V){
  float L=vAux.x*.25,F=mod(vAux.y,16.),sd=vAux.z,door=vAux.w;bool dead=vAux.y>15.5;
  float u=vUV.x,v=vUV.y;const float fh=2.85;
  float pgrain=fbm2(vec2(u*.42,v*.57)+vec2(sd*.071,sd*.023),3);
  float rainStain=pow(vnoise(vec2(u*.17+sd*.031,v*.045)),2.8);
  base*=mix(.84,1.12,pgrain)*(1.-rainStain*.13);
#ifdef KB_ARCH_HIGH
  float chip=fbm2(vec2(u*2.7+sd*.21,v*3.8),3);
  float stone=fbm2(vec2(u*.9+sd*.032,v*.38),3);
  float damp=(1.-smoothstep(.22,1.85,v))*(.35+.65*stone);
  float faded=smoothstep(.43,.76,chip)*.14;
  base*=1.-damp*.18-faded;
  base=mix(base,base*vec3(.78,.84,.72),smoothstep(.68,.86,stone)*damp*.18);
  rough=clamp(rough+.06*damp,0.,1.);
#endif
  // Kamena/betonska sokla u prizemlju sa okapnicom
  float pl=1.-smoothstep(.42,.68,v);
  if(v<.74&&v>.63){base=mix(base,vec3(.56,.54,.51),.45);rough=.78;}
  base=mix(base,base*vec3(.60,.58,.55),pl*.78);
  if(v>.62&&v<.74)Nm=normalize(N+vec3(0,.30,0));
  if(v<0.)base=mix(base,vec3(.26,.25,.24),.42);
  // Realistican udeo izlozene opeke (tipicne nedovrsene/ciglene kuce na Kalac brdu)
  if(mod(sd,5.5)<.95 && v>.66){
    vec2 brick=vec2(u*1.58+mod(floor(v*3.4),2.)*.5,v*3.4);
    vec2 seam=abs(fract(brick)-.5);
    float mortar=smoothstep(.44,.485,max(seam.x,seam.y));
    float tone=.84+.28*h21(floor(brick));
#ifdef KB_ARCH_HIGH
    base=mix(base,base*vec3(.86,.62,.52)*tone,(1.-mortar)*.38);
    base*=1.-mortar*.22;
#else
    base=mix(base,base*vec3(.86,.62,.52)*tone,mortar*.29);
#endif
  }
  if(v<0.||L<2.2)return;
  float fi=floor(v/fh),fv=v-fi*fh;
  if(fi>=F)return;
  // Medjuspratni venac i krovna streha-senka
  if(fi>0.&&fv<.17){base*=.72;if(fv<.05)base*=.85;}
  if(fi==F-1.&&fv>2.56){float es=smoothstep(2.56,2.84,fv);base*=mix(1.,.64,es);}
  float nC=max(1.,floor((L-1.1)/2.6+.5)),pitch=L/nC;
  float ci=clamp(floor(u/pitch),0.,nC-1.),cu=(ci+.5)*pitch,du=u-cu;
  float ww=min(1.28,pitch*.56),hw=ww*.5,hh=.68,wy=fv-1.5;
  float hs=h21(vec2(sd*3.7+fi*17.,ci*7.3+3.));
  vec3 frameC=sd<96.?vec3(.88,.86,.82):(sd<190.?vec3(.28,.18,.11):vec3(.48,.52,.55));
  // Ulazna vrata sa pragom, nadsvetlom i diskretnim svetlom iznad ulaza nocu
  if(door>.5&&fi<.5&&ci==floor(nC*.5)){
    float ax=abs(du);
    // Malo ulazno svetlo iznad vrata na naseljenim kucama
    if(!dead&&uDusk>.15&&hs<.55){
      float lp=length(vec2(du,fv-2.34));
      if(lp<.09){emis=vec3(1.,.82,.52)*3.8*uDusk;base=vec3(.9,.8,.6);}
      else if(lp<1.1){base+=vec3(1.,.78,.45)*exp(-lp*2.2)*.18*uDusk;}
    }
    if(ax<.62&&fv<2.20){
      float inner=step(ax,.46)*step(fv,2.06)*step(.07,fv);
      vec3 dc=sd<110.?vec3(.16,.09,.05):(sd<200.?vec3(.09,.18,.13):vec3(.16,.18,.21));
      // Paneli vrata
      float panel=step(abs(du),.34)*step(abs(fv-.62),.34)+step(abs(du),.34)*step(abs(fv-1.55),.38);
      base=mix(frameC*.85,dc*(1.-panel*.14),inner);rough=.52;spec=.06;
      if(fv<.08)base=vec3(.44,.43,.40);
      if(inner>.5){
        float pane=step(abs(du),.18)*step(1.38,fv)*step(fv,1.94);
        if(pane>.5){base=vec3(.02,.03,.038);glass=1.;rough=.10;spec=.62;emis=vec3(1.,.72,.38)*1.15*uDusk*step(.45,hs);}
        float knob=step(length(vec2(du-.33,fv-1.0)),.042);base=mix(base,vec3(.78,.68,.36),knob);
      }
      Nm=normalize(N+vec3(0,.26*sign(fv-1.05)*(1.-smoothstep(.008,.045,abs(abs(fv-1.05)-.59))),0));
    }
    return;
  }
  float ax=abs(du),ay=abs(wy);
  // Trag kise ispod prozorske klupcice
  if(ax<hw+.06&&wy<-hh-.10&&wy>-hh-.68){
    float st=(1.-(1.-smoothstep(-hh-.68,-hh-.12,wy)))*(1.-smoothstep(hw-.10,hw+.06,ax));
    base*=1.-st*.11;
  }
#ifdef KB_ARCH_HIGH
  if(ax<hw+.23&&wy>hh+.045&&wy<hh+.16){
    float ledge=1.-smoothstep(hh+.06,hh+.16,wy);
    base=mix(base,vec3(.42,.40,.37),ledge*.52);
    rough=.86;
  }
  if(ax<hw+.05&&ay<hh+.04){
    float opening=(1.-smoothstep(hw-.12,hw+.045,ax))*(1.-smoothstep(hh-.12,hh+.035,ay));
    base*=mix(1.,.68,opening*.28);
  }
#endif
  // Spoljni okvir i duboka senka spaletne
  if(ax<hw+.19&&ay<hh+.19){
    float edge=max(ax-hw,ay-hh);
    if(edge>0.){float lit=smoothstep(.02,.16,edge);base=mix(base,vec3(.62,.61,.57),.26+.24*lit);rough=.82;Nm=normalize(N+normalize(cross(vec3(0,1,0),N))*sign(du)*step(ay-hh,ax-hw)*.45+vec3(0,sign(wy)*step(ax-hw,ay-hh)*.45,0));}
    if(wy>hh&&wy<hh+.11)base*=.78;
  }
  // Izbacena prozorska klupcica sa senkom ispod
  if(ay<hh+.13&&ax<hw+.15&&wy<-hh&&wy>-hh-.13){base=vec3(.68,.66,.63)*(wy<-hh-.06?.64:1.03);rough=.72;Nm=normalize(N+vec3(0,.55,0));return;}
  // Roletne / kapci
  float blinds=h21(vec2(sd*1.13+fi,ci*3.9));
  if(blinds>.82&&ax<hw+.13&&ay<hh-.06&&ax>hw-.13){
    float slat=.78+.22*step(.25,fract((wy+hh)*10.5));
    base=vec3(.34,.24,.15)*slat;Nm=normalize(N+vec3(0,.22*cos((wy+hh)*66.),0));rough=.78;return;
  }
  if(ax<hw&&ay<hh){
    Nm=N;
    bool fr=ax>hw-.085||ay>hh-.085;
    bool bar=ax<.026||(abs(wy-.12)<.028&&hs<.72);
    if(fr||bar){vec3 tangent=normalize(cross(vec3(0,1,0),N));
      Nm=normalize(N+tangent*(ax/hw>ay/hh?sign(du)*.32:0.)+vec3(0,ay/hh>=ax/hw?sign(wy)*.32:0.,0));base=frameC;rough=.46;spec=.07;return;}
    glass=1.;rough=.08;spec=.68;
    float glow;vec3 room=interiorRoom(vec2(du,wy),vec2(hw,hh),N,V,hs,glow);
    float on=(!dead&&hs<.52)?1.:0.;vec3 lamp=hs>.43?vec3(.88,.94,1.):vec3(1.,.68,.37);
    base=room*(.48-.28*uDusk); emis=room*lamp*(.06+on*glow*2.5*uDusk);
    glass=mix(.68,.30,on*uDusk);rough=.14;
  }
}

void main(){
  if(vT>9.1&&vT<9.4&&leafCoverage(vUV)<.5)discard;
  if(vT>7.0&&vT<8.0){oC=vec4(vC*1.6,vT-7.0);return;}

  float wetMask=0.;
  bool isSidewalk=(cls==5.&&vT>5.15&&vT<5.35);
  bool isAsphalt=(cls==5.&&!isSidewalk&&dot(vC,vec3(.3,.59,.11))<.42);
  if(uTexOn>.5&&(cls<7.||cls==9.)){
    float L=-1.,sc=1.;
    float lum=dot(vC,vec3(.3,.59,.11));
    if(cls==1.){float fr=vT-cls;L=fr>.05?1.:(fr<-.05?0.:(h31(floor(vC*255.))>.55?1.:0.));sc=L>.5?.52:.26;rough=.88;}
    else if(cls==2.){L=2.;sc=.78;rough=.60;spec=.06;}
    else if(cls==3.){bool leaf=vC.g>vC.r*1.12&&vC.g>vC.b*1.1;L=leaf?5.:3.;sc=leaf?.8:.72;rough=.88;}
    else if(cls==5.){

      else{L=lum<.42?4.:(vC.r>vC.g*.95&&lum>.45?6.:7.);sc=L==4.?.26:(L==6.?.72:.42);rough=L==4.?.68:.92;wetMask=L==4.?1.:.65;}

    else if(cls==6.){float pn=vnoise(vP.xz*.045);L=pn>.72?7.:5.;sc=L==7.?.36:.92;rough=.92;}
    else if(cls==9.){L=5.;sc=1.45;rough=.76;spec=.05;}
    if(L>=0.){
      vec3 nn;float rm;vec4 t=tex1(L,vP,N,w,sc,nn,rm);
      base*=t.rgb*1.92;Nm=nn;texRM=rm;
      rough=clamp(rough+(rm-.5)*.26,.08,.98);
    }
  }
  // Mikro-detalji asfalta i betonskih trotoara/ivicnjaka
  if(isAsphalt){
    float macro=fbm2(vP.xz*.09,3);
    float patchN=smoothstep(.64,.82,vnoise(vP.xz*.18+3.1));
    float lane=abs(vUV.x),ruts=vAux.x>.5?exp(-pow((lane-vAux.x*.021)*3.,2.)):0.;
    base*=1.-ruts*.13;rough-=ruts*.09;
    base*=mix(.86,1.10,macro)*(1.-patchN*.12);
    rough=clamp(rough-(1.-macro)*.10,.25,.95);
  }else if(isSidewalk){
    vec2 tile=fract(vP.xz*.65);
    float joint=1.-smoothstep(.015,.045,min(min(tile.x,1.-tile.x),min(tile.y,1.-tile.y)));
    base*=1.-joint*.22;
    rough=clamp(rough+joint*.12,.3,.95);
  }
  if(cls==1.&&vAux.x>0.)facade(base,emis,glass,rough,spec,Nm,N,V);
#ifdef KB_ARCH_HIGH
  if(cls==2.){
    vec2 tile=vP.xz*vec2(1.08,2.10);
    tile.x+=mod(floor(tile.y),2.)*.50;
    vec2 tf=fract(tile),tb=floor(tile);
    float seam=1.-smoothstep(.022,.092,min(min(tf.x,1.-tf.x),min(tf.y,1.-tf.y)));

  }
  if(cls==1.&&glass<.5&&vAux.x>0.){
    float patina=fbm2(vP.xz*.28+vP.yy*.05,3);
    base*=1.-.075*smoothstep(.55,.78,patina);
  }
#endif
#ifdef KB_ARCH_MAX
  if(cls==2.){
    float moss=fbm2(vP.xz*.38+vec2(23.1),3);

#endif

  }

  // Blic / munja
  {float fd=length(vP-uFlashPos);col+=base*vec3(.78,.91,1.)*(1.-smoothstep(1.,34.,fd))*uFlash*3.2;}

  // Fresnel refleksija neba na mokrom asfaltu, barama i staklu
  if((uWet>.01&&cls==5.)||glass>.5){

    float fr=pow(clamp(1.-max(dot(Nm,V),0.),0.,1.),4.);
    float k=(glass>.5?.65:uWet*(.35+.65*pud))*(.10+.90*fr)*(1.-rough);

  }

  col+=emis;

  // Atmosferska magla: daljinska + dolinska (Kalac brdo padine) + mesecina u magli

  col=mix(col,fogCol,f);
  oC=vec4(col,isAsphalt?pud*uWet:0.);
}`;

   }

const FS_COMP=(def)=>GLSL_HEAD+def+`

  vec2 d=(uv-.5)*uCA*(.35+length(uv-.5)*1.5);
  vec3 c;
#ifdef CA
  c=vec3(texture(uHDR,uv+d).r,texture(uHDR,uv).g,texture(uHDR,uv-d).b);
#else
  c=texture(uHDR,uv).rgb;
#endif

  c+=texture(uBloom,uv).rgb*uBloomK;
#endif

  c*=mix(vec3(1.),vec3(1.07,1.0,.90),uCine*uNight*smoothstep(.45,1.,l));

  c=max(c-.006*uCine*uNight,0.);

}`;

/* FXAA + Contrast-Adaptive Sharpening (CAS) za kristalnu ostrinu bez zamucenja */
const FS_FXAA=GLSL_HEAD+`in vec2 uv;layout(location=0) out vec4 oC;uniform sampler2D uSrc;uniform vec2 uTexel;uniform float uSharp;

function chunkize(src,CS,SX){
  SX=SX||7;
  const N=(src.length/SX)|0,T=(N/3)|0;
  if(!T)return {buf:new ArrayBuffer(0),chunks:[],verts:0};
  let bx0=1e9,bz0=1e9,bx1=-1e9,bz1=-1e9;const cx=new Float32Array(T),cz=new Float32Array(T);
  for(let t=0,k=0;t<T;t++,k+=3*SX){const x=(src[k]+src[k+SX]+src[k+2*SX])/3,z=(src[k+2]+src[k+SX+2]+src[k+2*SX+2])/3;cx[t]=x;cz[t]=z;if(x<bx0)bx0=x;if(x>bx1)bx1=x;if(z<bz0)bz0=z;if(z>bz1)bz1=z}
  const gx=Math.max(1,Math.floor((bx1-bx0)/CS)+1),gz=Math.max(1,Math.floor((bz1-bz0)/CS)+1),cells=gx*gz,cell=new Int32Array(T),count=new Int32Array(cells);
  for(let t=0;t<T;t++){const ci=Math.min(gz-1,Math.floor((cz[t]-bz0)/CS))*gx+Math.min(gx-1,Math.floor((cx[t]-bx0)/CS));cell[t]=ci;count[ci]++}
  const start=new Int32Array(cells+1);for(let i=0;i<cells;i++)start[i+1]=start[i]+count[i];
  const fill=start.slice(0,cells),buf=new ArrayBuffer(N*28),f=new Float32Array(buf),u=new Uint32Array(buf),h=new Int16Array(buf),b=new Uint8Array(buf);
  const mn=new Float32Array(cells*3).fill(1e9),mx=new Float32Array(cells*3).fill(-1e9);
  const q16=v=>Math.max(-32767,Math.min(32767,Math.round(v*16)));
  for(let t=0;t<T;t++){const ci=cell[t],dst=fill[ci]++;
    for(let v=0;v<3;v++){const s=(t*3+v)*SX,iv=dst*3+v,d=iv*7,x=src[s],y=src[s+1],z=src[s+2];f[d]=x;f[d+1]=y;f[d+2]=z;u[d+3]=packRGB(src[s+3],src[s+4],src[s+5]);f[d+4]=src[s+6];
      if(SX>=13){h[iv*14+10]=q16(src[s+7]);h[iv*14+11]=q16(src[s+8]);b[iv*28+24]=src[s+9];b[iv*28+25]=src[s+10];b[iv*28+26]=src[s+11];b[iv*28+27]=src[s+12]}
      const m=ci*3;if(x<mn[m])mn[m]=x;if(x>mx[m])mx[m]=x;if(y<mn[m+1])mn[m+1]=y;if(y>mx[m+1])mx[m+1]=y;if(z<mn[m+2])mn[m+2]=z;if(z>mx[m+2])mx[m+2]=z}}
  const chunks=[];
  for(let ci=0;ci<cells;ci++){if(!count[ci])continue;const m=ci*3,a=[mn[m],mn[m+1],mn[m+2]],b2=[mx[m],mx[m+1],mx[m+2]],c=[(a[0]+b2[0])/2,(a[1]+b2[1])/2,(a[2]+b2[2])/2];
    chunks.push({first:start[ci]*3,count:count[ci]*3,min:a,max:b2,c,r:Math.hypot(b2[0]-a[0],b2[1]-a[1],b2[2]-a[2])/2})}
  return {buf,chunks,verts:N};
}
