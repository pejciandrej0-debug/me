function nearGrass(p){
  if(interior||quality<2||season===3||!r2Cache)return null;
  const cx=Math.floor(p.x/8),cz=Math.floor(p.z/8),key=cx+','+cz+','+quality+','+season;
  if(key===grassPatchKey)return grassPatch;
  const step=[0,0,2.5,2.05,1.75][quality],reach=[0,0,42,49,56][quality],points=[];
  for(let iz=Math.floor((cz*8-reach)/step);iz<=Math.ceil((cz*8+reach+8)/step);iz++)
    for(let ix=Math.floor((cx*8-reach)/step);ix<=Math.ceil((cx*8+reach+8)/step);ix++){
      const h='grass'+ix+','+iz,x=(ix+.15+.7*r2Hash(h,0))*step,z=(iz+.15+.7*r2Hash(h,1))*step;
      if(Math.abs(x)>HALF-2||Math.abs(z)>HALF-2||Math.abs(x-FIELD.x)<FIELD.w/2+6.7&&Math.abs(z-FIELD.z)<FIELD.d/2+6.7)continue;
      if(r2ZoneAt(x,z)<.91)continue;
      let excluded=false;
      if(pointInside(x,z,PARK.poly))for(const path of PARK.paths)for(let j=1;j<path.length;j++)if(segDist(x,z,path[j-1],path[j])<1.65)excluded=true;
      if(excluded)continue;
      points.push(x,height(x,z)-.07,z,.66+.62*r2Hash(h,2),r2Hash(h,3)*TAU,r2Hash(h,4),r2Hash(h,5),5);
    }
  grassPatchKey=key;grassPatch=new Float32Array(points);return grassPatch;
}
function r2VegBuild(){
  const q=clamp(quality|0,0,4),dens=[.5,.75,1,1.3,1.6][q],out=[];r2BushSpots=[];
  const put=(x,z,sc,kind,key)=>{out.push(x,height(x,z),z,sc,r2Hash(key,3)*6.283,r2Hash(key,5),r2Hash(key,7),kind)};
  trees.forEach((t,i)=>{const kind=(t.pine&&!t.park)?0:((r2Hash('sp'+i,19)<.28)?3:(r2Hash('sp'+i,20)<.42?4:1));put(t.x,t.z,t.park?7.8:t.h,kind,'t'+i)});
  const free=(x,z,pad)=>{if(Math.abs(x)>HALF-3||Math.abs(z)>HALF-3)return false;if(buildingAt(x,z,pad))return false;const r=nearestRoad(x,z);if(r.d<(r.road?r.road.width/2:3)+1.1)return false;if(hyp(x-FIELD.x,z-FIELD.z)<40&&Math.abs(x-FIELD.x)<36&&Math.abs(z-FIELD.z)<27)return false;return true};
  // podrast po šumama: mreža 20 m sa deterministički pomerenim tačkama
  for(let gx=-HALF;gx<HALF;gx+=20)for(let gz=-HALF;gz<HALF;gz+=20){const k='u'+gx+','+gz,x=gx+r2Hash(k,1)*20,z=gz+r2Hash(k,2)*20;
    if(r2MaskAt(x,z)>140&&r2Hash(k,4)<.62*dens&&free(x,z,1.4)){put(x,z,1.0+r2Hash(k,6)*1.3,2,k);if(r2Hash(k,8)<.12)r2BushSpots.push({x,z})}}
  // dvorišta: žbunje i voćke oko kuća
  houses.forEach((h,i)=>{const sp=houseSpec(h),k='g'+(h.sourceId||i);if(sp.floors>3||sp.area<30||sp.area>400||r2Hash(k,1)>.5*dens)return;
    const rad=Math.sqrt(sp.area)*.5+2.5,n=1+(r2Hash(k,2)<.5?1:0);
    for(let j=0;j<n;j++){const a=r2Hash(k+j,3)*6.283,d=rad+r2Hash(k+j,4)*3.5,x=h.c[0]+Math.cos(a)*d,z=h.c[1]+Math.sin(a)*d;if(!free(x,z,1.2))continue;
      if(j===0&&r2Hash(k,5)<.35)put(x,z,3.2+r2Hash(k,6)*1.6,1,k+'f');else put(x,z,.9+r2Hash(k+j,7)*1.1,2,k+j)}});
  return new Float32Array(out);
}
