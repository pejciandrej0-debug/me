        master.gain.setTargetAtTime(muted()||(H.paused&&!H.mini)||document.hidden?0:.75,t,.035);
        if (listener.positionX) {
          for (const [key,value] of Object.entries({positionX:p.x,positionY:y,positionZ:p.z,
            forwardX:Math.sin(H.yaw),forwardY:0,forwardZ:Math.cos(H.yaw),upX:0,upY:1,upZ:0})) listener[key].setValueAtTime(value,t);
        } else { listener.setPosition(p.x,y,p.z); listener.setOrientation(Math.sin(H.yaw),0,Math.cos(H.yaw),0,1,0); }
        const station=H.locations[8], blocked=!H.lineClear(p,{x:station.x,z:station.z+4},.1);
        const volume=H.paused||H.interior||H.football?0:clamp(1-distance(p,station)/65,0,1);
        hum.forEach((n,i)=>{n.gain.gain.setTargetAtTime(volume*(i?.025:.055),t,.15);
          n.filter.frequency.setTargetAtTime(blocked?320:1800,t,.15);});
        if (motor) {
          const speed=Math.abs(H.ride.speed), on=H.ride.kind===2&&H.ride.battery>0&&!H.interior&&!H.football&&!H.paused;
          position(motor.spatial,p,.45);
          motor.oscillator.frequency.setTargetAtTime(70+speed*28,t,.06);
          motor.gain.gain.setTargetAtTime(on&&speed>.1?.012+Math.min(.045,speed*.003):0,t,.08);
        }
      }
